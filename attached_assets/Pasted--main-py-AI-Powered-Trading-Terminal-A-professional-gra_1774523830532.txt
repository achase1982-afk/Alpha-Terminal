# HELPER – simple markdown-to-HTML (avoids injection, keeps formatting)

# =========================================================================

def _markdown_to_html(text: str) -> str:
“””
Minimal markdown-to-HTML conversion for the AI response box.
Handles headers, bold, code, and line breaks.
“””
import re
text = text.replace(”&”, “&”).replace(”<”, “<”).replace(”>”, “>”)
text = re.sub(r’^### (.+)$’, r’<h3>\1</h3>’, text, flags=re.MULTILINE)
text = re.sub(r’^## (.+)$’, r’<h2>\1</h2>’, text, flags=re.MULTILINE)
text = re.sub(r’^# (.+)$’, r’<h1>\1</h1>’, text, flags=re.MULTILINE)
text = re.sub(r’**(.+?)**’, r’<strong>\1</strong>’, text)
text = re.sub(r’`(.+?)`’, r’<code>\1</code>’, text)
text = text.replace(”\n”, “<br>”)
return text

if st.button("Send to AI", use_container_width=True):
    if not user_question.strip():

                user_question=user_question,

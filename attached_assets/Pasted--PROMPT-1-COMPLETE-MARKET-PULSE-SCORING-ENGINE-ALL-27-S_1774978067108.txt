function scoreRates(data: MarketData): ClusterResult {
  let score = 0;
  const rules: string[] = [];

  const tnxChange = getPercentChange('$TNX');
  if (tnxChange !== null) {
    if (tnxChange <= -1.0) { score += 1.0; rules.push('TNX down >1%: +1.0'); }
    else if (tnxChange <= -0.5) { score += 0.75; rules.push('TNX down 0.5-1%: +0.75'); }
    else if (tnxChange <= -0.1) { score += 0.5; rules.push('TNX down 0.1-0.5%: +0.5'); }
    else if (tnxChange >= 1.0) { score -= 1.0; rules.push('TNX up >1%: -1.0'); }
    else if (tnxChange >= 0.5) { score -= 0.75; rules.push('TNX up 0.5-1%: -0.75'); }
    else if (tnxChange >= 0.1) { score -= 0.5; rules.push('TNX up 0.1-0.5%: -0.5'); }
  }

  // TYX directional confirmation/divergence
  const tyxChange = getPercentChange('$TYX');
  if (tyxChange !== null && tnxChange !== null) {
    const sameDir = (tnxChange > 0 && tyxChange > 0) || (tnxChange < 0 && tyxChange < 0);
    if (!sameDir && Math.abs(tnxChange) > 0.1 && Math.abs(tyxChange) > 0.1) {
      score -= 0.25;
      rules.push('TNX/TYX divergence: -0.25');
    }
  }

  // Bond futures confirmation
  const zbChange = getPercentChange('/ZB');
  const ztChange = getPercentChange('/ZT');
  if (zbChange !== null && ztChange !== null) {
    const bondsRallying = zbChange > 0 && ztChange > 0;
    const bondsSelling = zbChange < 0 && ztChange < 0;

    if (score > 0 && bondsRallying) {
      score += 0.25;
      rules.push('Bond futures confirm yield decline: +0.25');
    } else if (score > 0 && bondsSelling) {
      score -= 0.25;
      rules.push('Bond futures contradict yield decline: -0.25');
    } else if (score < 0 && bondsSelling) {
      score -= 0.25;
      rules.push('Bond futures confirm yield rise: -0.25');
    } else if (score < 0 && bondsRallying) {
      score += 0.25;
      rules.push('Bond futures contradict yield rise: +0.25');
    }
  }

  return { name: 'rates', score: clamp(score, -2.0, 2.0), rules };
}

    else if (vixChange <= -3) { score += 0.75; rules.push('VIX down 3-5% intraday: +0.75'); }
    else if (vixChange >= 5) { score -= 1.5; rules.push('VIX up >5% intraday: -1.5'); }

  // VVIX (vol of vol)
  const vvixChange = getPercentChange('VVIX');
  if (vvixChange !== null) {
    if (vvixChange <= -3) { score += 0.5; rules.push('VVIX down >3%: +0.5'); }
    else if (vvixChange >= 3) { score -= 0.5; rules.push('VVIX up >3%: -0.5'); }

function scoreBreadth(data: MarketData): ClusterResult {
  let score = 0;
  const rules: string[] = [];

  const advn = getValue('$ADVN');
  const decn = getValue('$DECN');
  const trin = getValue('$TRIN');
  const tick = getValue('$TICK');
  const add = getValue('$ADD');

  // AD Ratio
  if (advn !== null && decn !== null && decn > 0) {
    const adRatio = advn / decn;
    if (adRatio > 3) { score += 2.0; rules.push(`AD ratio ${adRatio.toFixed(2)} (>3): +2.0`); }
    else if (adRatio > 2) { score += 1.5; rules.push(`AD ratio ${adRatio.toFixed(2)} (2-3): +1.5`); }
    else if (adRatio > 1.5) { score += 0.75; rules.push(`AD ratio ${adRatio.toFixed(2)} (1.5-2): +0.75`); }
    else if (adRatio < 0.33) { score -= 2.0; rules.push(`AD ratio ${adRatio.toFixed(2)} (<0.33): -2.0`); }
    else if (adRatio < 0.5) { score -= 1.5; rules.push(`AD ratio ${adRatio.toFixed(2)} (0.33-0.5): -1.5`); }
    else if (adRatio < 0.67) { score -= 0.75; rules.push(`AD ratio ${adRatio.toFixed(2)} (0.5-0.67): -0.75`); }
  }

  // TRIN tiered scoring
  if (trin !== null) {
    if (trin < 0.5) { score += 1.0; rules.push(`TRIN ${trin.toFixed(4)} (<0.5 strongly bullish vol breadth): +1.0`); }
    else if (trin < 1.0) { score += 0.5; rules.push(`TRIN ${trin.toFixed(4)} (0.5-1.0 bullish vol breadth): +0.5`); }
    else if (trin >= 1.5 && trin < 2.0) { score -= 0.5; rules.push(`TRIN ${trin.toFixed(4)} (1.5-2.0 mildly bearish): -0.5`); }
    else if (trin >= 2.0 && trin < 2.5) { score -= 1.0; rules.push(`TRIN ${trin.toFixed(4)} (2.0-2.5 bearish): -1.0`); }
    else if (trin >= 2.5 && trin < 3.0) { score -= 1.5; rules.push(`TRIN ${trin.toFixed(4)} (2.5-3.0 very bearish): -1.5`); }
    else if (trin >= 3.0) { score -= 2.0; rules.push(`TRIN ${trin.toFixed(4)} (>=3.0 extreme bearish): -2.0`); }
  }

  // TRIN cluster cap: if TRIN >= 3.0, breadth cannot be positive
  if (trin !== null && trin >= 3.0 && score > 0) {
    score = 0;
    rules.push('TRIN >=3.0 cluster cap: score clamped to 0');
  }

  return { name: 'breadth', score: clamp(score, -3.0, 3.0), rules };
}

function scoreRiskAppetite(data: MarketData): ClusterResult {
  let score = 0;
  const rules: string[] = [];

  const esChange = getPercentChange('/ES');
  const nqChange = getPercentChange('/NQ');
  const ymChange = getPercentChange('/YM');
  const rtyChange = getPercentChange('/RTY');

  const changes = [esChange, nqChange, rtyChange, ymChange].filter(c => c !== null) as number[];
  if (changes.length < 2) return { name: 'riskAppetite', score: 0, rules: ['Insufficient data'] };

  const allPositive = changes.every(c => c > 0);
  const allNegative = changes.every(c => c < 0);
  const avg = changes.reduce((a, b) => a + b, 0) / changes.length;

  // Broad rally/selloff strength
  if (allPositive && avg > 1.0) { score += 1.5; rules.push(`All futures up, avg +${avg.toFixed(2)}%: +1.5`); }
  else if (allPositive && avg > 0.5) { score += 1.0; rules.push(`All futures up, avg +${avg.toFixed(2)}%: +1.0`); }
  else if (allPositive) { score += 0.5; rules.push(`All futures up, avg +${avg.toFixed(2)}%: +0.5`); }
  else if (allNegative && avg < -1.0) { score -= 1.5; rules.push(`All futures down, avg ${avg.toFixed(2)}%: -1.5`); }
  else if (allNegative && avg < -0.5) { score -= 1.0; rules.push(`All futures down, avg ${avg.toFixed(2)}%: -1.0`); }
  else if (allNegative) { score -= 0.5; rules.push(`All futures down, avg ${avg.toFixed(2)}%: -0.5`); }

  // Leadership quality: Russell vs S&P
  if (rtyChange !== null && esChange !== null) {
    const rtyVsEs = rtyChange - esChange;
    if (rtyVsEs > 0.5) { score += 0.5; rules.push(`RTY outperforming ES by ${rtyVsEs.toFixed(2)}%: +0.5`); }
    else if (rtyVsEs < -0.5) { score -= 0.5; rules.push(`RTY underperforming ES by ${Math.abs(rtyVsEs).toFixed(2)}%: -0.5`); }
  }

  // Narrow tech leadership warning
  if (nqChange !== null && esChange !== null && rtyChange !== null && ymChange !== null) {
    const nqVsOthers = nqChange - ((esChange + rtyChange + ymChange) / 3);
    if (nqVsOthers > 1.0) { score -= 0.25; rules.push(`NQ leading others by ${nqVsOthers.toFixed(2)}% (narrow): -0.25`); }
  }

  return { name: 'riskAppetite', score: clamp(score, -2.0, 2.0), rules };
}

## COMPOSITE, CONFIDENCE, AND BIAS LABEL

```typescript
function calculateComposite(clusters: ClusterResult[]): PulseOutput {
  // Weighted composite
  let composite = 0;
  for (const cluster of clusters) {
    const weight = CLUSTER_WEIGHTS[cluster.name] / 100;
    composite += cluster.score * weight;
  }

  // Base confidence from composite magnitude (0-100 scale)
  let confidence = Math.min(100, Math.round(Math.abs(composite) * 40));

  // Cluster disagreement penalty
  const scores = clusters.map(c => c.score);
  const positiveCount = scores.filter(s => s > 0).length;
  const negativeCount = scores.filter(s => s < 0).length;
  const disagreementPairs = Math.min(positiveCount, negativeCount);
  const disagreementPenalty = disagreementPairs * 10;
  confidence = Math.max(0, confidence - disagreementPenalty);

  // Bias label
  let label: string;
  if (composite > 1.75 && confidence >= 75) label = 'STRONGLY BULLISH';
  else if (composite > 0.75) label = 'MODERATELY BULLISH';
  else if (composite > 0.25) label = 'SLIGHTLY BULLISH';
  else if (composite >= -0.25) label = 'NEUTRAL';
  else if (composite >= -0.75) label = 'SLIGHTLY BEARISH';
  else if (composite >= -1.75) label = 'MODERATELY BEARISH';
  else if (composite < -1.75 && confidence >= 75) label = 'STRONGLY BEARISH';
  else label = 'NEUTRAL';

  // Today's edge (for the Strategist)
  let todayEdge: string;
  if (confidence < 40) todayEdge = 'NO_EDGE';
  else if (composite > 0.5) todayEdge = 'BULLISH_EDGE';
  else if (composite < -0.5) todayEdge = 'BEARISH_EDGE';
  else todayEdge = 'NEUTRAL_EDGE';

  // Size recommendation
  let size: string;
  if (confidence < 40) size = 'NO_TRADE';
  else if (confidence < 60) size = 'HALF_SIZE';
  else size = 'FULL_SIZE';

  return {
    composite: Math.round(composite * 100) / 100,
    confidence,
    label,
    todayEdge,
    size,
    clusters,
    weights: CLUSTER_WEIGHTS,
    timestamp: Date.now(),
  };
}

# PROMPT 2: FIX OPTIONS STRATEGIST – SHOW REAL STRIKES FROM SCHWAB

The Options Strategist currently generates strategy recommendations but does NOT show specific strikes, specific expiration dates, or exact premium amounts from the Schwab options chain. It’s giving vague output like “Net Credit $2.90 per spread” without telling the user which exact contracts to trade.

Fix this by having the backend pull the real options chain from Schwab and select specific strikes before sending anything to Gemini.

-----

## THE PROBLEM

Right now the Strategist flow is:

1. User taps “Strategist”
1. Frontend sends the current symbol + pulse data to Gemini
1. Gemini guesses at strikes, premiums, and expiration dates
1. User gets a recommendation with made-up numbers

This is wrong. Gemini must never calculate options math.

-----

## THE CORRECT FLOW

```
User taps Strategist
        |
        v
Backend receives symbol + pulse output (bias, edge, confidence)
        |
        v
Backend calls Schwab GET /chains endpoint for that symbol
(with appropriate parameters: contractType, strikeCount, range, expMonth)
        |
        v
Backend deterministic engine selects optimal strikes:
  - Reads delta, bid, ask, volume, open interest from chain
  - Filters for liquid contracts (volume > 10, OI > 100, bid/ask spread < 30% of mid)
  - Selects strikes based on pulse bias and strategy type
  - Calculates max profit, max loss, POP from the real chain data
        |
        v
Backend packages a STRICT JSON payload with all solved math
        |
        v
Gemini receives JSON and ONLY generates the narrative explanation
Gemini cannot change any number in the payload
        |
        v
Frontend displays the trade card with real strikes and the AI explanation
```

## SCHWAB OPTIONS CHAIN ENDPOINT

The Schwab API endpoint for options chains is:

```
GET /marketdata/v1/chains?symbol={symbol}&contractType={type}&strikeCount={count}&range=OTM&expMonth={month}
```

You already have Schwab OAuth working. Use the existing authenticated client.

The response includes for each contract:

- putExpDateMap or callExpDateMap (nested by expiration date then strike)
- For each contract: bid, ask, last, mark, delta, gamma, theta, vega, volume, openInterest, strikePrice, expirationDate, daysToExpiration, inTheMoney

function selectStrategies(edge: string, symbol: string, chain: SchwabChainResponse): StrategyPayload[] {
  const strategies: StrategyPayload[] = [];

  if (edge === 'BEARISH_EDGE') {
    // Strategy 1: Bear Call Spread (call credit spread)
    strategies.push(buildBearCallSpread(chain, symbol));
    // Strategy 2: Bear Put Spread (put debit spread) for higher conviction
    strategies.push(buildBearPutSpread(chain, symbol));
  } else if (edge === 'BULLISH_EDGE') {
    // Strategy 1: Bull Put Spread (put credit spread)
    strategies.push(buildBullPutSpread(chain, symbol));
    // Strategy 2: Bull Call Spread (call debit spread)
    strategies.push(buildBullCallSpread(chain, symbol));
  } else if (edge === 'NEUTRAL_EDGE') {
    // Strategy 1: Iron Condor
    strategies.push(buildIronCondor(chain, symbol));
    // Strategy 2: Short Strangle (if account allows)
    strategies.push(buildShortStrangle(chain, symbol));
  } else {
    // NO_EDGE: no strategies, tell user to sit out
    return [];
  }

  return strategies;
}
```

## EXAMPLE: BUILD BEAR CALL SPREAD

```typescript
function buildBearCallSpread(chain: SchwabChainResponse, symbol: string): StrategyPayload {
  // Get the nearest weekly or next Friday expiration with 1-7 DTE
  const targetDTE = findExpiration(chain.callExpDateMap, { minDTE: 1, maxDTE: 7 });

  // Get all call strikes for that expiration
  const calls = getCallsForExpiration(chain, targetDTE);

  // Filter for liquidity
  const liquidCalls = calls.filter(c =>
    c.volume > 10 &&
    c.openInterest > 100 &&
    (c.ask - c.bid) / ((c.ask + c.bid) / 2) < 0.30
  );

  // Short strike: closest to 0.20 delta (OTM, ~80% POP)
  const shortLeg = liquidCalls.reduce((best, c) =>
    Math.abs(Math.abs(c.delta) - 0.20) < Math.abs(Math.abs(best.delta) - 0.20) ? c : best
  );

  // Long strike: next strike above short (typically $5 wide for SPX, $1-2 for equities)
  // Find the next available strike above the short strike
  const longLeg = liquidCalls
    .filter(c => c.strikePrice > shortLeg.strikePrice)
    .sort((a, b) => a.strikePrice - b.strikePrice)[0];

  if (!longLeg) throw new Error('Cannot construct spread: no long strike available');

  const netCredit = (shortLeg.bid + shortLeg.ask) / 2 - (longLeg.bid + longLeg.ask) / 2;
  const strikeWidth = longLeg.strikePrice - shortLeg.strikePrice;
  const maxProfit = netCredit * 100; // per contract
  const maxLoss = (strikeWidth - netCredit) * 100;
  const pop = Math.round((1 - Math.abs(shortLeg.delta)) * 100);
  const breakeven = shortLeg.strikePrice + netCredit;

  return {
    strategy_type: 'Bear Call Spread (Call Credit Spread)',
    expiration_date: targetDTE.expirationDate,
    days_to_expiration: targetDTE.daysToExpiration,
    short_leg: {
      strike: shortLeg.strikePrice,
      type: 'CALL',
      action: 'SELL',
      bid: shortLeg.bid,
      ask: shortLeg.ask,
      mark: (shortLeg.bid + shortLeg.ask) / 2,
      delta: shortLeg.delta,
      volume: shortLeg.volume,
      openInterest: shortLeg.openInterest,
    },
    long_leg: {
      strike: longLeg.strikePrice,
      type: 'CALL',
      action: 'BUY',
      bid: longLeg.bid,
      ask: longLeg.ask,
      mark: (longLeg.bid + longLeg.ask) / 2,
      delta: longLeg.delta,
      volume: longLeg.volume,
      openInterest: longLeg.openInterest,
    },
    net_credit: Math.round(netCredit * 100) / 100,
    max_profit: Math.round(maxProfit * 100) / 100,
    max_loss: Math.round(maxLoss * 100) / 100,
    breakeven: Math.round(breakeven * 100) / 100,
    probability_of_profit_pct: pop,
    risk_reward_ratio: `1:${(maxLoss / maxProfit).toFixed(2)}`,
    size_recommendation: 'Based on $250 max risk per trade',
    contracts: Math.max(1, Math.floor(250 / maxLoss)),
    exit_rules: {
      profit_target_pct: 50,
      profit_target_amount: Math.round(maxProfit * 0.5 * 100) / 100,
      stop_loss_pct: 100,
      stop_loss_amount: maxProfit, // 2x credit = 100% of credit received
      time_exit: 'Close by expiration day if not exited',
    },
  };
}
```

Apply the same pattern for buildBullPutSpread, buildBullCallSpread, buildBearPutSpread, buildIronCondor, buildShortStrangle. Each one:

1. Reads the real chain from Schwab
1. Filters for liquidity
1. Selects strikes based on delta targets
1. Calculates all math deterministically
1. Packages into the StrategyPayload

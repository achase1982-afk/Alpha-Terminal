import { useState, useEffect, useRef, useCallback } from "react";
import { Sidebar } from "@/components/Sidebar";
import { MetricsBar } from "@/components/MetricsBar";
import { TradingChart } from "@/components/TradingChart";
import { OptionsTab } from "@/components/OptionsTab";
import { AiIntelligenceTab } from "@/components/AiIntelligenceTab";
import { MarketScanner } from "@/components/MarketScanner";
import { TickerTape } from "@/components/TickerTape";
import { TickerSearch } from "@/components/TickerSearch";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { useTerminalStore } from "@/lib/store";
import { NewsTab } from "@/components/NewsTab";
import { AiBiasStrip } from "@/components/market-pulse/AiBiasStrip";
import { 
  LineChart, BarChart2, BrainCircuit, Menu, Radar, 
  Newspaper, Activity, Briefcase, ListOrdered, Building2 
} from "lucide-react";

export default function TerminalPage() {
  const { symbol, streamStatus } = useTerminalStore();
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [isScrolled, setIsScrolled] = useState(false);
  const [activeMainTab, setActiveMainTab] = useState("news");
  const scrollRef = useRef<HTMLDivElement>(null);

  // Collapsed Header Logic
  const handleScroll = useCallback(() => {
    const el = scrollRef.current;
    if (!el) return;
    setIsScrolled(el.scrollTop > 60);
  }, []);

  return (
    <div className="app-shell bg-background h-screen flex flex-col overflow-hidden select-none">
      
      {/* --- HEADER BLOCK --- */}
      <header className="shrink-0 bg-background z-50 border-b border-card-border">
        <div className="flex items-center h-12 px-4 bg-card">
          <button onClick={() => setSidebarOpen(true)} className="p-2 mr-2 text-muted-foreground hover:text-primary">
            <Menu className="w-5 h-5" />
          </button>
          <div className="flex flex-col leading-none">
            <span className="font-black text-base tracking-wider">ALPHA</span>
            <span className="font-semibold text-[9px] tracking-[0.25em] text-primary">TERMINAL</span>
          </div>
          <div className="ml-auto flex items-center gap-2">
            <span className="font-mono text-xs text-primary font-bold">{symbol}</span>
            <div className={`w-2 h-2 rounded-full ${streamStatus === 'live' ? 'bg-terminal-success shadow-[0_0_8px_#00d166]' : 'bg-terminal-danger'}`} />
          </div>
        </div>
        <TickerTape />
      </header>

      <div className="flex flex-1 min-h-0 relative">
        
        {/* --- FULL-SCREEN SIDEBAR MODAL --- */}
        <Sidebar isOpen={sidebarOpen} onClose={() => setSidebarOpen(false)} />

        <main ref={scrollRef} onScroll={handleScroll} className="flex-1 overflow-y-auto app-content pb-24 scrollbar-hide">
          
          {/* STICKY ACTION BAR: Collapses from Metrics to a thin Buy/Sell strip */}
          <div className="sticky top-0 z-40 bg-background border-b border-card-border/50">
             <AiBiasStrip onNavigateToPulse={() => setActiveMainTab("ai")} />
             <MetricsBar compact={isScrolled} />
          </div>

          <TickerSearch />

          <Tabs value={activeMainTab} onValueChange={setActiveMainTab} className="w-full">
            {/* STATIC TOP MENU: Spelled out, 4-column grid */}
            <div className="sticky top-[108px] z-30 bg-background/95 backdrop-blur-md py-2 px-2">
              <TabsList className="grid grid-cols-4 w-full bg-card/50 border border-card-border h-10 p-1">
                <TabsTrigger value="news" className="text-[10px] font-bold">NEWS</TabsTrigger>
                <TabsTrigger value="options" className="text-[10px] font-bold">OPTIONS</TabsTrigger>
                <TabsTrigger value="chart" className="text-[10px] font-bold">CHART</TabsTrigger>
                <TabsTrigger value="company" className="text-[10px] font-bold">COMPANY</TabsTrigger>
              </TabsList>
            </div>

            <TabsContent value="news" className="m-0"><NewsTab /></TabsContent>
            <TabsContent value="options" className="m-0"><OptionsTab /></TabsContent>
            <TabsContent value="chart" className="m-0 h-[500px]"><TradingChart symbol={symbol} /></TabsContent>
            <TabsContent value="company" className="m-0"><CompanyResearchHub /></TabsContent>
          </Tabs>
        </main>

        {/* --- PERSISTENT BOTTOM NAV --- */}
        <nav className="fixed bottom-0 left-0 right-0 h-16 bg-[#0c0c0c] border-t border-primary/20 flex items-center justify-around px-2 z-50 backdrop-blur-lg">
          <NavBtn icon={<Radar />} label="SCANNER" />
          <NavBtn icon={<Activity />} label="MARKETS" />
          
          {/* THE AI HUB (Pulse + Strategist + Chat) */}
          <button className="relative -top-5 w-14 h-14 bg-primary rounded-full flex items-center justify-center shadow-[0_0_20px_rgba(255,184,0,0.3)] text-black active:scale-95 transition-transform">
            <BrainCircuit className="w-7 h-7" />
          </button>

          <NavBtn icon={<Briefcase />} label="PORTFOLIO" />
          <NavBtn icon={<ListOrdered />} label="WATCHLIST" />
        </nav>
      </div>
    </div>
  );
}

function NavBtn({ icon, label }: { icon: React.ReactNode, label: string }) {
  return (
    <button className="flex flex-col items-center gap-1 text-muted-foreground hover:text-primary transition-colors">
      {React.cloneElement(icon as React.ReactElement, { className: "w-5 h-5" })}
      <span className="text-[8px] font-bold tracking-tighter">{label}</span>
    </button>
  );
}

function Badge({ children, color = gold }) {
return (
<span style={{
background: color + “18”,
border: `1px solid ${color}55`,
color: color,
fontSize: 9,
fontFamily: “monospace”,
letterSpacing: “0.12em”,
padding: “2px 7px”,
borderRadius: 3,
textTransform: “uppercase”,

}}>{children}</span>

}

function SectionHeader({ label, onToggle, open }) {
return (
<div
onClick={onToggle}
style={{
display: “flex”,
justifyContent: “space-between”,
alignItems: “center”,
cursor: “pointer”,
paddingBottom: 8,
marginBottom: 10,
borderBottom: `1px solid ${cardBorder}`,
userSelect: “none”,
}}
>
<span style={{
fontSize: 10,
fontFamily: “monospace”,
letterSpacing: “0.18em”,
color: gold,
fontWeight: 700,
textTransform: “uppercase”,
}}>{label}</span>
<span style={{ color: textMuted, fontSize: 11 }}>{open ? “▲” : “▼”}</span>
</div>
);
}

function Section({ label, children, defaultOpen = true }) {
const [open, setOpen] = useState(defaultOpen);
return (
<div style={{ marginBottom: 18 }}>
<SectionHeader label={label} onToggle={() => setOpen(o => !o)} open={open} />
{open && children}
</div>
);
}

function MetricGrid({ items }) {
return (
<div style={{
display: “grid”,
gridTemplateColumns: “1fr 1fr”,
gap: “10px 16px”,
marginBottom: 4,
}}>
{items.map(([label, value, sub], i) => (
<div key={i}>
<div style={{ fontSize: 9, color: textMuted, letterSpacing: “0.1em”, textTransform: “uppercase”, marginBottom: 2, fontFamily: “monospace” }}>{label}</div>
<div style={{ fontSize: 15, fontWeight: 600, color: textPrimary, fontFamily: “monospace”, lineHeight: 1.2 }}>{value ?? neu(”—”)}</div>
{sub && <div style={{ fontSize: 9, color: textSub, marginTop: 1, fontFamily: “monospace” }}>{sub}</div>}
</div>
))}
</div>
);
}

function MetricRow({ label, value, highlight }) {
return (

}}>

<span style={{ fontSize: 12, fontWeight: 600, color: highlight || textPrimary, fontFamily: “monospace” }}>{value}</span>
</div>
);
}

}

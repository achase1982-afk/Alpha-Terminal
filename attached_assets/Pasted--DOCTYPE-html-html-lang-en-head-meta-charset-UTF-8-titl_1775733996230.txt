  <script>
    function showScreen(type) {
      const opt = document.getElementById('screen-options');
      const stk = document.getElementById('screen-stock');
      const tabs = document.querySelectorAll('.btn-tab');
      if (type === 'options') {
        opt.classList.add('active');
        stk.classList.remove('active');
        tabs[0].classList.add('active');
        tabs[1].classList.remove('active');
      } else {
        stk.classList.add('active');
        opt.classList.remove('active');
        tabs[1].classList.add('active');
        tabs[0].classList.remove('active');
      }
    }

    function toggleRiskDetails() {
      const panel = document.getElementById('risk-details');
      const chev = document.getElementById('risk-chevron');
      const isOpen = panel.style.display === 'block';
      panel.style.display = isOpen ? 'none' : 'block';
      chev.innerHTML = isOpen ? '&#9662;' : '&#9652;';
    }
  </script>
